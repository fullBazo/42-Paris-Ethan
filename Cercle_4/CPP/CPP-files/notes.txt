STDIO STREAMS

std::cout = sortie standard;

std::endl = retour a la ligne;

<< = redirection;

ex : std::cout << "Hello World !" << std::endl;
sortie : Hello World !\n$

std::cin = entree standard (peut etre utilisee pour attendre un input utilisateur)

>> redirection;

ex : 
std::cout << "Input a word: ";
std::cin >> buff;
std::cout << "You entered: [" << buff << "]" << std::endl;
